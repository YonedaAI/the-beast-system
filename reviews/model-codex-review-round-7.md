   176	    """A hyperarc (T, h) generated when every premise in ``requires`` holds.
   177	
   178	    h needs an output of AT LEAST ONE member of T (provider(s) -> dependent). ``incidence`` is
   179	    the incidence Inc(k, m) the rule instantiates (the problem of Ax_k arises about pillar m's
   180	    inputs or rules, giving arc pillar(k) -> m), or None for a rule that is not an incidence.
   181	    ``core`` marks membership of D*; non-core rules are reported, not needed.
   182	    """
   183	
   184	    tail: frozenset[Pillar]
   185	    head: Pillar
   186	    premise: str
   187	    requires: frozenset[AnyAxiom]
   188	    source: str
   189	    core: bool
   190	    incidence: tuple[Axiom, Pillar] | None = None
   191	
   192	    def __post_init__(self) -> None:
   193	        tail, _ = arc_key(self.tail, self.head)
   194	        requires = frozenset(self.requires)
   195	        if not requires or any(not isinstance(a, (Axiom, IncidenceAxiom)) for a in requires):
   196	            raise ValueError("a dependency rule needs at least one axiom premise")
   197	        if self.incidence is not None:
   198	            axiom, target = self.incidence
   199	            if tail != frozenset({axiom.pillar}) or target is not self.head:
   200	                raise ValueError("an incidence Inc(k, m) generates exactly pillar(k) -> m")
   201	            if axiom not in requires:
   202	                raise ValueError("an incidence rule must require its own axiom")
   203	        object.__setattr__(self, "tail", tail)
   204	        object.__setattr__(self, "requires", requires)
   205	
   206	    @property
   207	    def key(self) -> ArcKey:
   208	        return self.tail, self.head
   209	
   210	
   211	def _inc(axiom: Axiom, target: Pillar, premise: str, requires: Iterable[AnyAxiom],
   212	         source: str, core: bool = True) -> DependencyRule:
   213	    return DependencyRule(frozenset({axiom.pillar}), target, premise,
   214	                          frozenset(requires) | {axiom}, source, core, (axiom, target))

   268	def rules_holding(
   269	    axioms: Iterable[AnyAxiom], rules: Iterable[DependencyRule] = CORE_RULES
   270	) -> tuple[DependencyRule, ...]:
   271	    """The CORE rules whose premises all hold under ``axioms``.
   272	
   273	    Non-core (further) rules are never generated, whatever ``axioms`` contains.
   274	    """
   275	    held = frozenset(axioms)
   276	    return tuple(r for r in rules if r.core and r.requires <= held)

   316	@dataclass(frozen=True)
   317	class GovernanceSystem:
   318	    """A collective that makes and enforces collective decisions (KB Section 4.1).
   319	
   320	    ``axioms`` lists the axioms that hold of the collective; ``persistent`` means it survives
   321	    over the period studied; it may include the incidence axioms X1, X2, X3, X5. Both are
   322	    coded independently of ``practices``. Dependencies are generated from ``axioms``;
   323	    ``absent_incidences`` records incidences Inc(k, m) OBSERVED not to arise in this system,
   324	    which suppress the arcs they would generate (the route by which a case can falsify D0 or
   325	    interdependence).
   326	    """
   327	
   328	    name: str
   329	    persistent: bool
   330	    axioms: frozenset[AnyAxiom]
   331	    practices: tuple[Practice, ...] = ()
   332	    absent_incidences: frozenset[tuple[Axiom, Pillar]] = frozenset()
   333	    note: str = ""
   334	
   335	    def __post_init__(self) -> None:
   336	        if not isinstance(self.persistent, bool):
   337	            raise TypeError("persistent must be a bool")
   338	        axioms = frozenset(self.axioms)
   339	        if any(not isinstance(a, (Axiom, IncidenceAxiom)) for a in axioms):
   340	            raise TypeError("axioms must be Axiom or IncidenceAxiom members")
   341	        practices = tuple(self.practices)
   342	        if any(not isinstance(p, Practice) for p in practices):
   343	            raise TypeError("practices must be Practice instances")
   344	        object.__setattr__(self, "axioms", axioms)
   345	        object.__setattr__(self, "practices", practices)
   346	        absent = frozenset(tuple(i) for i in self.absent_incidences)
   347	        if any(len(i) != 2 or not isinstance(i[0], Axiom) or not isinstance(i[1], Pillar)
   348	               for i in absent):
   349	            raise TypeError("absent_incidences must be (Axiom, Pillar) pairs")
   350	        object.__setattr__(self, "absent_incidences", absent)
   351	
   352	    def holds(self, axiom: AnyAxiom) -> bool:
   353	        """Predicate for one axiom (Ax1..Ax6 or an incidence axiom X_k)."""
   354	        return axiom in self.axioms
   355	
   356	    @property
   357	    def all_axioms_hold(self) -> bool:
   358	        """Ax1..Ax6 all hold (the antecedent of Theorem A)."""
   359	        return all(self.holds(a) for a in Axiom)
   360	
   361	    @property
   362	    def extended_axioms_hold(self) -> bool:
   363	        """Ax1..Ax6 and X1, X2, X3, X5 all hold (the antecedent of interdependence)."""
   364	        return EXTENDED_AXIOMS <= self.axioms

   367	def generate_dependencies(
   368	    system: GovernanceSystem, rules: Iterable[DependencyRule] = CORE_RULES
   369	) -> frozenset[ArcKey]:
   370	    """Hyperarcs generated in ``system``: rules whose premises hold, minus absent incidences."""
   371	    return frozenset(
   372	        r.key for r in rules_holding(system.axioms, rules)
   373	        if r.incidence is None or r.incidence not in system.absent_incidences
   374	    )

    74	class CodedRecord:
    75	    """The coded record of a governance system: practices plus index inputs.
    76	
    77	    ``shares`` may omit pillars with no holders. Each share vector must be supported on the
    78	    performers of that pillar's practices, so F2 holds for every image by construction and
    79	    the theorem's content rests on F1 and F4. ``coalitions`` maps a coalition name to its
    80	    kappa_m(S); None gives singleton coalitions with kappa_m({v}) = s^m_v.
    81	    """
    82	
    83	    system: GovernanceSystem
    84	    shares: Mapping[Pillar, Mapping[str, float]]
    85	    contestability: Mapping[Pillar, float]
    86	    weights: Mapping[Pillar, float] | None = None
    87	    coalitions: Mapping[str, Mapping[Pillar, float]] | None = None
    88	    links: tuple[CaptureLink, ...] = ()

    90	    def __post_init__(self) -> None:
    91	        roles = role_map(self.system)
    92	        for m, s in self.shares.items():
    93	            total = sum(float(v) for v in s.values())
    94	            if any(float(v) < 0 for v in s.values()) or abs(total - 1.0) > _ATOL:
    95	                raise ValueError(f"shares[{m.name}] must be a probability vector")
    96	            for actor, v in s.items():
    97	                if float(v) > 0 and m not in roles.get(actor, frozenset()):
    98	                    raise ValueError(f"{actor!r} has share in {m.name} but performs no practice")
    99	        for m in PILLARS:
   100	            validate_unit_interval(self.contestability[m], f"Q[{m.name}]")

   132	def represent(record: CodedRecord) -> FormalizedStructure:
   133	    """The profile map pi: G -> F_G.
   134	
   135	    V = practice performers plus actors named in shares or links (roles come only from
   136	    practices). m in rho(v) iff v performs a practice of pillar m and m is implemented.
   137	    Shares, Q, weights, coalitions and links are copied; D = arcs generated from G's axioms.
   138	    Layers: a^m_ij = s^m_i for i != j, so out-strength shares reproduce s^m.
   139	    """
   140	    system = record.system
   141	    implemented = {m for m in PILLARS if realized_form(system, m) is not None}
   142	    performers = role_map(system)
   143	    named = {a for s in record.shares.values() for a in s}
   144	    named |= {e for link in record.links for e in (link.source, link.target)}
   145	    actors = tuple(sorted(set(performers) | named))
   146	    roles = {
   147	        v: frozenset(m.name for m in performers.get(v, frozenset()) if m in implemented)
   148	        for v in actors
   149	    }
   150	    shares = {
   151	        m.name: {v: float(record.shares.get(m, {}).get(v, 0.0)) for v in actors}
   152	        for m in PILLARS
   153	    }
   154	    idx = {v: k for k, v in enumerate(actors)}
   155	    layers = {}
   156	    for m in M_LABELS:
   157	        arr = np.zeros((len(actors), len(actors)))
   158	        for v, s in shares[m].items():
   159	            arr[idx[v], :] = s
   160	        np.fill_diagonal(arr, 0.0)
   161	        layers[m] = arr
   162	    weights = record.weights or {m: 1.0 / len(PILLARS) for m in PILLARS}
   163	    if record.coalitions is None:
   164	        kappa = {f"{{{v}}}": {m: shares[m][v] for m in M_LABELS} for v in actors}
   165	    else:
   166	        kappa = {name: {m.name: float(row.get(m, 0.0)) for m in PILLARS}
   167	                 for name, row in record.coalitions.items()}
   168	    links = frozenset((lk.source, lk.target, lk.pillar.name) for lk in record.links if lk.holds)
   169	    arcs = frozenset(
   170	        (frozenset(t.name for t in tail), head.name)
   171	        for tail, head in generate_dependencies(system)
   172	    )
   173	    return FormalizedStructure(
   174	        actors=actors, pillars=M_LABELS, roles=roles, layers=layers, shares=shares,
   175	        contestability={m.name: float(record.contestability[m]) for m in PILLARS},
   176	        weights={m.name: float(weights[m]) for m in PILLARS},
   177	        kappa=kappa, links=links, arcs=arcs,
   178	    )

   192	def _closure(arcs: Iterable[LabelArc], start: Iterable[Label]) -> frozenset[Label]:
   193	    """cl(S) on string labels (same rule as necessity.cascade_closure)."""
   194	    arcs = list(arcs)
   195	    closed = set(start)
   196	    changed = True
   197	    while changed:
   198	        changed = False
   199	        for tail, head in arcs:
   200	            if head not in closed and tail <= closed:
   201	                closed.add(head)
   202	                changed = True
   203	    return frozenset(closed)

   206	def _cascade_complete(arcs: Iterable[LabelArc], labels: tuple[Label, ...]) -> bool:
   207	    keep = frozenset(labels)
   208	    inside = [(t, h) for t, h in arcs if h in keep and t <= keep]
   209	    return bool(keep) and all(_closure(inside, {n}) >= keep for n in keep)

   212	@dataclass(frozen=True)
   213	class ClassMembership:
   214	    """Conditions F1..F4 of the formalized class."""
   215	
   216	    f1: bool
   217	    f2: bool
   218	    f3: bool
   219	    f4: bool
   220	
   221	    @property
   222	    def holds(self) -> bool:
   223	        return self.f1 and self.f2 and self.f3 and self.f4

   225	
   226	def in_formalized_class(F: FormalizedStructure) -> ClassMembership:
   227	    """F1 every layer has a holder; F2 s^m a probability vector on holders of m, a^m_ij > 0
   228	    only if i holds m; F3 Q, omega, kappa in range, link endpoints in V; F4 D contains D0 and
   229	    is strongly connected and cascade-complete."""
   230	    labels_ok = F.pillars == M_LABELS
   231	    f1 = labels_ok and all(F.holders(m) for m in F.pillars)
   232	    f2 = labels_ok
   233	    for k, m in enumerate(F.pillars):
   234	        s = np.array([F.shares[m].get(v, 0.0) for v in F.actors])
   235	        holders = F.holders(m)
   236	        on_holders = all(F.shares[m].get(v, 0.0) == 0.0 or v in holders for v in F.actors)
   237	        rows_ok = all(not F.layers[m][i].any() or v in holders for i, v in enumerate(F.actors))
   238	        if s.min(initial=0.0) < 0 or abs(s.sum() - 1.0) > _ATOL or not on_holders or not rows_ok:
   239	            f2 = False
   240	    vals = [*F.contestability.values(), *F.weights.values(),
   241	            *(x for row in F.kappa.values() for x in row.values())]
   242	    f3 = (
   243	        all(math.isfinite(x) and 0.0 <= x <= 1.0 for x in vals)
   244	        and abs(sum(F.weights.values()) - 1.0) <= _ATOL
   245	        and all(a in F.actors and b in F.actors for a, b, _ in F.links)
   246	    )
   247	    f4 = (
   248	        labels_ok
   249	        and D0_LABELS <= F.arcs
   250	        and is_strongly_connected(_underlying(F.arcs, F.pillars), set(F.pillars))
   251	        and _cascade_complete(F.arcs, F.pillars)
   252	    )
   253	    return ClassMembership(f1, f2, f3, f4)

   266	    H: dict[Label, float]
   267	    U: float

   274	def preserved_quantities(
   275	    F: FormalizedStructure, thresholds: CaptureThresholds = CaptureThresholds()
   276	) -> PreservedQuantities:
   277	    """H_m, U, C_mn (coalition overlap), I, Cap (Proposition B) and Inter on F's own labels."""
   278	    H = {m: hhi([F.shares[m].get(v, 0.0) for v in F.actors]) if F.holders(m) else 0.0
   279	         for m in F.pillars}
   280	    U = sum(F.weights[m] * H[m] * (1.0 - F.contestability[m]) for m in F.pillars)
   281	    pairs = list(itertools.combinations(F.pillars, 2))
   282	    C = {frozenset((m, n)): max((min(row.get(m, 0.0), row.get(n, 0.0))
   283	                                 for row in F.kappa.values()), default=0.0)
   284	         for m, n in pairs}

   286	    cap = U >= thresholds.u_star and I >= thresholds.i_star and bool(F.links)
   287	    inter = (is_strongly_connected(_underlying(F.arcs, F.pillars), set(F.pillars))
   288	             and _cascade_complete(F.arcs, F.pillars))
   289	    return PreservedQuantities(H, float(U), C, float(I), cap, inter)

   298	    return abs(a - b) <= _ATOL

   301	def equivalent_A(F: FormalizedStructure, G: FormalizedStructure) -> bool:  # noqa: N802
   302	    """F and G agree on roles, shares, Q, omega, kappa, links and D under an actor bijection.
   303	
   304	    The bijection is searched among actors with identical (roles, shares) signatures. kappa is
   305	    compared as a multiset of coalition rows (coalition names are labels, not actors).
   306	    """
   307	    if F.pillars != G.pillars or len(F.actors) != len(G.actors) or F.arcs != G.arcs:
   308	        return False
   309	    for m in F.pillars:
   310	        if not _same(F.contestability[m], G.contestability[m]):
   311	            return False
   312	        if not _same(F.weights[m], G.weights[m]):
   313	            return False
   314	    rows = lambda X: sorted(tuple(round(r.get(m, 0.0), 12) for m in X.pillars)  # noqa: E731
   315	                            for r in X.kappa.values())
   316	    if rows(F) != rows(G):
   317	        return False
   318	    groups: dict[tuple, list[str]] = {}
   319	    for v in G.actors:
   320	        groups.setdefault(_signature(G, v), []).append(v)
   321	    f_groups: dict[tuple, list[str]] = {}
   322	    for v in F.actors:
   323	        f_groups.setdefault(_signature(F, v), []).append(v)
   324	    if {k: len(v) for k, v in groups.items()} != {k: len(v) for k, v in f_groups.items()}:
   325	        return False
   326	    keys = list(f_groups)
   327	    # Links are the only remaining constraint; try permutations within signature groups.
   328	    for perms in itertools.product(*(itertools.permutations(groups[k]) for k in keys)):
   329	        phi = {u: w for k, perm in zip(keys, perms) for u, w in zip(f_groups[k], perm)}
   330	        if {(phi[a], phi[b], m) for a, b, m in F.links} == set(G.links):
   331	            return True
   332	    return False

   338	def drop(F: FormalizedStructure, m: Label) -> FormalizedStructure:
   339	    """Delete pillar m: its layer, shares, Q, kappa column and every hyperarc involving m.
   340	
   341	    Remaining weights are renormalized to sum 1.
   342	    """
   343	    if m not in F.pillars:
   344	        raise ValueError(f"unknown pillar {m!r}")
   345	    keep = tuple(p for p in F.pillars if p != m)
   346	    total = sum(F.weights[p] for p in keep)
   347	    weights = {p: (F.weights[p] / total if total > 0 else 1.0 / len(keep)) for p in keep}
   348	    return FormalizedStructure(
   349	        actors=F.actors, pillars=keep,
   350	        roles={v: ps - {m} for v, ps in F.roles.items()},
   351	        layers={p: F.layers[p] for p in keep},
   352	        shares={p: dict(F.shares[p]) for p in keep},
   353	        contestability={p: F.contestability[p] for p in keep},
   354	        weights=weights,
   355	        kappa={s: {p: row.get(p, 0.0) for p in keep} for s, row in F.kappa.items()},
   356	        links=frozenset(lk for lk in F.links if lk[2] != m),
   357	        arcs=frozenset((t, h) for t, h in F.arcs if h != m and m not in t),
   358	    )

   361	def merge(F: FormalizedStructure, m: Label, n: Label, lam: float) -> FormalizedStructure:
   362	    """Replace m, n by one pillar 'm+n': holders united, shares, layers, Q and kappa mixed with
   363	    weight lam on m; omega_m + omega_n; arcs relabeled, self-loops removed."""
   364	    if m == n or m not in F.pillars or n not in F.pillars:
   365	        raise ValueError("merge needs two distinct existing pillars")
   366	    lam = validate_unit_interval(lam, "lam")
   367	    new = f"{m}+{n}"
   368	    keep = tuple(new if p == m else p for p in F.pillars if p != n)
   369	    relabel = lambda p: new if p in (m, n) else p  # noqa: E731
   370	
   371	    def mix(a: float, b: float) -> float:
   372	        return lam * a + (1.0 - lam) * b
   373	
   374	    shares = {p: dict(F.shares[p]) for p in keep if p != new}
   375	    shares[new] = {v: mix(F.shares[m].get(v, 0.0), F.shares[n].get(v, 0.0)) for v in F.actors}
   376	    layers = {p: F.layers[p] for p in keep if p != new}
   377	    layers[new] = lam * F.layers[m] + (1.0 - lam) * F.layers[n]
   378	    contest = {p: F.contestability[p] for p in keep if p != new}
   379	    contest[new] = mix(F.contestability[m], F.contestability[n])
   380	    weights = {p: F.weights[p] for p in keep if p != new}
   381	    weights[new] = F.weights[m] + F.weights[n]
   382	    kappa = {}
   383	    for s, row in F.kappa.items():
   384	        r = {p: row.get(p, 0.0) for p in keep if p != new}
   385	        r[new] = mix(row.get(m, 0.0), row.get(n, 0.0))
   386	        kappa[s] = r
   387	    arcs = set()
   388	    for t, h in F.arcs:
   389	        t2, h2 = frozenset(relabel(x) for x in t), relabel(h)
   390	        if h2 not in t2:
   391	            arcs.add((t2, h2))
   392	    return FormalizedStructure(
   393	        actors=F.actors, pillars=keep,
   394	        roles={v: frozenset(relabel(p) for p in ps) for v, ps in F.roles.items()},
   395	        layers=layers, shares=shares, contestability=contest, weights=weights, kappa=kappa,
   396	        links=frozenset((a, b, relabel(p)) for a, b, p in F.links),
   397	        arcs=frozenset(arcs),
   398	    )

   116	    seen = {start}

   119	        for nxt in graph.get(stack.pop(), frozenset()):

   126	def is_strongly_connected(graph: Graph, nodes: Iterable[Pillar] | None = None) -> bool:
   127	    """True iff every node of the graph is reachable from every other node.
   128	
   129	    ``nodes`` restricts the graph to an induced subgraph (default: all nodes that appear).
   130	    The empty graph is not strongly connected; a single node is.
   131	    """
   132	    keep = set(nodes) if nodes is not None else _nodes(graph)
   133	    if not keep:
   134	        return False
   135	    fwd: dict[Pillar, frozenset[Pillar]] = {
   136	        u: frozenset(v for v in graph.get(u, ()) if v in keep) for u in keep
   137	    }
   138	    rev: dict[Pillar, set[Pillar]] = {u: set() for u in keep}
   139	    for u, vs in fwd.items():
   140	        for v in vs:
   141	            rev[v].add(u)
   142	    start = next(iter(keep))
   143	    rev_f = {u: frozenset(v) for u, v in rev.items()}
   144	    return _reach(fwd, start) == keep and _reach(rev_f, start) == keep

   149	    nodes = _nodes(graph)
